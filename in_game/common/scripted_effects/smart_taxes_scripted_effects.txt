		add_to_variable_list = { name = smart_taxes_enabled_estates target = estate_type:cossacks_estate }
		add_to_variable_list = { name = smart_taxes_enabled_estates target = estate_type:dhimmi_estate }
		if = {
			limit = { NOT = { exists = var:auto_coin_minting } }
			# Vanilla's on value for auto minting: in_game/common/scripted_guis/economy_satisfaction_target.txt:266 (toggle_auto_coin_minting)
