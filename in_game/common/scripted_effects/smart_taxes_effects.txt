# Toggle smart taxes for a single estate. Adds or removes the estate from the enabled list.
# Scope: country. Requires scope:estate_type.
smart_taxes_toggle_estate = {
	if = {
		limit = {
			is_target_in_variable_list = { name = smart_taxes_enabled_estates target = scope:estate_type }
		}
		remove_list_variable = { name = smart_taxes_enabled_estates target = scope:estate_type }
	} else = {
		add_to_variable_list = { name = smart_taxes_enabled_estates target = scope:estate_type }
		set_variable = { name = smart_taxes_recalculate value = yes }
	}
}

# Suppress "estate_type is used but never set" warning.
_smart_taxes_suppress = {
	if = { limit = { always = no }
		save_scope_as = estate_type
	}
}
