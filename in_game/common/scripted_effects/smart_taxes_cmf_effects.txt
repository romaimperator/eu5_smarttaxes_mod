# Sync all CMM settings to country variables.
# Scope: country
smart_taxes_sync_all = {
	# Enabled toggles → inverted alias (enabled=1 removes _disabled var)
	cmm_sync_bool_alias_inverted = { setting = smart_taxes__nobles_enabled alias = nobles_estate_disabled }
	cmm_sync_bool_alias_inverted = { setting = smart_taxes__clergy_enabled alias = clergy_estate_disabled }
	cmm_sync_bool_alias_inverted = { setting = smart_taxes__burghers_enabled alias = burghers_estate_disabled }
	cmm_sync_bool_alias_inverted = { setting = smart_taxes__peasants_enabled alias = peasants_estate_disabled }
	cmm_sync_bool_alias_inverted = { setting = smart_taxes__tribes_enabled alias = tribes_estate_disabled }
	cmm_sync_bool_alias_inverted = { setting = smart_taxes__cossacks_enabled alias = cossacks_estate_disabled }
	cmm_sync_bool_alias_inverted = { setting = smart_taxes__dhimmi_enabled alias = dhimmi_estate_disabled }

	# Target sliders → copy raw 0–100 value via alias, then scale to 0.0–1.0
	cmm_sync_setting_alias = { setting = smart_taxes__nobles_target alias = nobles_estate_target }
	set_variable = { name = nobles_estate_target value = { value = var:nobles_estate_target multiply = 0.01 } }

	cmm_sync_setting_alias = { setting = smart_taxes__clergy_target alias = clergy_estate_target }
	set_variable = { name = clergy_estate_target value = { value = var:clergy_estate_target multiply = 0.01 } }

	cmm_sync_setting_alias = { setting = smart_taxes__burghers_target alias = burghers_estate_target }
	set_variable = { name = burghers_estate_target value = { value = var:burghers_estate_target multiply = 0.01 } }

	cmm_sync_setting_alias = { setting = smart_taxes__peasants_target alias = peasants_estate_target }
	set_variable = { name = peasants_estate_target value = { value = var:peasants_estate_target multiply = 0.01 } }

	cmm_sync_setting_alias = { setting = smart_taxes__tribes_target alias = tribes_estate_target }
	set_variable = { name = tribes_estate_target value = { value = var:tribes_estate_target multiply = 0.01 } }

	cmm_sync_setting_alias = { setting = smart_taxes__cossacks_target alias = cossacks_estate_target }
	set_variable = { name = cossacks_estate_target value = { value = var:cossacks_estate_target multiply = 0.01 } }

	cmm_sync_setting_alias = { setting = smart_taxes__dhimmi_target alias = dhimmi_estate_target }
	set_variable = { name = dhimmi_estate_target value = { value = var:dhimmi_estate_target multiply = 0.01 } }

	set_variable = { name = smart_taxes_recalculate value = yes }
}

# Callback handler for cmf_on_callback.
# var:cmf_callback is the flag of the setting that was changed.
# Scope: country

		trigger = var:cmf_callback
		# Enabled toggles
		flag:smart_taxes__nobles_enabled = {
