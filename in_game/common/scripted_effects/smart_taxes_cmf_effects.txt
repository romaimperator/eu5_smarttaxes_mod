# Sync all CMM settings to country variables.
# Scope: country
smart_taxes_sync_all = {
	# Enabled toggles → inverted alias (enabled=1 removes _disabled var)
	cmm_sync_bool_alias_inverted = { setting = smart_taxes__nobles_enabled alias = nobles_estate_disabled }
	cmm_sync_bool_alias_inverted = { setting = smart_taxes__clergy_enabled alias = clergy_estate_disabled }
	cmm_sync_bool_alias_inverted = { setting = smart_taxes__burghers_enabled alias = burghers_estate_disabled }
	cmm_sync_bool_alias_inverted = { setting = smart_taxes__peasants_enabled alias = peasants_estate_disabled }
	cmm_sync_bool_alias_inverted = { setting = smart_taxes__tribes_enabled alias = tribes_estate_disabled }
	cmm_sync_bool_alias_inverted = { setting = smart_taxes__cossacks_enabled alias = cossacks_estate_disabled }
	cmm_sync_bool_alias_inverted = { setting = smart_taxes__dhimmi_enabled alias = dhimmi_estate_disabled }

	# Target sliders → divide by 100 to convert 0–100 integer to 0.0–1.0 float
	set_variable = { name = nobles_estate_target value = "variable_map(cmm|flag:smart_taxes__nobles_target)" }
	change_variable = { name = nobles_estate_target divide = 100 }

	set_variable = { name = clergy_estate_target value = "variable_map(cmm|flag:smart_taxes__clergy_target)" }
	change_variable = { name = clergy_estate_target divide = 100 }

	set_variable = { name = burghers_estate_target value = "variable_map(cmm|flag:smart_taxes__burghers_target)" }
	change_variable = { name = burghers_estate_target divide = 100 }

	set_variable = { name = peasants_estate_target value = "variable_map(cmm|flag:smart_taxes__peasants_target)" }
	change_variable = { name = peasants_estate_target divide = 100 }

	set_variable = { name = tribes_estate_target value = "variable_map(cmm|flag:smart_taxes__tribes_target)" }
	change_variable = { name = tribes_estate_target divide = 100 }

	set_variable = { name = cossacks_estate_target value = "variable_map(cmm|flag:smart_taxes__cossacks_target)" }
	change_variable = { name = cossacks_estate_target divide = 100 }

	set_variable = { name = dhimmi_estate_target value = "variable_map(cmm|flag:smart_taxes__dhimmi_target)" }
	change_variable = { name = dhimmi_estate_target divide = 100 }

	set_variable = { name = smart_taxes_recalculate value = yes }
}

# Callback handler for cmf_on_callback.
# var:cmf_callback is the flag of the setting that was changed.
# Scope: country
smart_taxes_handle_callback = {
	switch = {
		trigger = var:cmf_callback
		# Enabled toggles
		flag:smart_taxes__nobles_enabled = {
			cmm_sync_bool_alias_inverted = { setting = smart_taxes__nobles_enabled alias = nobles_estate_disabled }
		}
		flag:smart_taxes__clergy_enabled = {
			cmm_sync_bool_alias_inverted = { setting = smart_taxes__clergy_enabled alias = clergy_estate_disabled }
		}
		flag:smart_taxes__burghers_enabled = {
			cmm_sync_bool_alias_inverted = { setting = smart_taxes__burghers_enabled alias = burghers_estate_disabled }
		}
		flag:smart_taxes__peasants_enabled = {
			cmm_sync_bool_alias_inverted = { setting = smart_taxes__peasants_enabled alias = peasants_estate_disabled }
		}
		flag:smart_taxes__tribes_enabled = {
			cmm_sync_bool_alias_inverted = { setting = smart_taxes__tribes_enabled alias = tribes_estate_disabled }
		}
		flag:smart_taxes__cossacks_enabled = {
			cmm_sync_bool_alias_inverted = { setting = smart_taxes__cossacks_enabled alias = cossacks_estate_disabled }
		}
		flag:smart_taxes__dhimmi_enabled = {
			cmm_sync_bool_alias_inverted = { setting = smart_taxes__dhimmi_enabled alias = dhimmi_estate_disabled }
		}
		# Target sliders
		flag:smart_taxes__nobles_target = {
			set_variable = { name = nobles_estate_target value = "variable_map(cmm|flag:smart_taxes__nobles_target)" }
			change_variable = { name = nobles_estate_target divide = 100 }
		}
		flag:smart_taxes__clergy_target = {
			set_variable = { name = clergy_estate_target value = "variable_map(cmm|flag:smart_taxes__clergy_target)" }
			change_variable = { name = clergy_estate_target divide = 100 }
		}
		flag:smart_taxes__burghers_target = {
			set_variable = { name = burghers_estate_target value = "variable_map(cmm|flag:smart_taxes__burghers_target)" }
			change_variable = { name = burghers_estate_target divide = 100 }
		}
		flag:smart_taxes__peasants_target = {
			set_variable = { name = peasants_estate_target value = "variable_map(cmm|flag:smart_taxes__peasants_target)" }
			change_variable = { name = peasants_estate_target divide = 100 }
		}
		flag:smart_taxes__tribes_target = {
			set_variable = { name = tribes_estate_target value = "variable_map(cmm|flag:smart_taxes__tribes_target)" }
			change_variable = { name = tribes_estate_target divide = 100 }
		}
		flag:smart_taxes__cossacks_target = {
			set_variable = { name = cossacks_estate_target value = "variable_map(cmm|flag:smart_taxes__cossacks_target)" }
			change_variable = { name = cossacks_estate_target divide = 100 }
		}
		flag:smart_taxes__dhimmi_target = {
			set_variable = { name = dhimmi_estate_target value = "variable_map(cmm|flag:smart_taxes__dhimmi_target)" }
			change_variable = { name = dhimmi_estate_target divide = 100 }
		}
	}
	set_variable = { name = smart_taxes_recalculate value = yes }
