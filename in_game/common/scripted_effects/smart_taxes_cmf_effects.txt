# Root scope: country.
smart_taxes_register_mod = {
	# Items: 1=Nobles, 2=Clergy, 3=Burghers, 4=Peasants, 5=Tribes, 6=Cossacks, 7=Dhimmi
	# Fields: f1=Enabled (bool), f2=Target (slider)
	cmm_register_settings_list = {
		mod_id = smart_taxes
		setting_id = estates
		tab_id = settings
		item_count = 7
		is_ordered = 0
	}

	cmm_register_list_bool_field = {
		mod_id = smart_taxes
		setting_id = estates
		field_id = enabled
		default_value = 0
	}

	cmm_register_list_slider_field = {
		mod_id = smart_taxes
		setting_id = estates
		field_id = target
		default_value = 50
		min_value = 0
		max_value = 100
		step_value = 1
	}

	cmm_set_list_field_conditional_format = { mod_id = smart_taxes setting_id = estates field_id = target }

	# Hide estates the player doesn't have, show ones they do
	if = { limit = { country_has_estate = estate_type:nobles_estate }
		cmm_show_list_item = { mod_id = smart_taxes setting_id = estates item = 1 }
	} else = {
		cmm_hide_list_item = { mod_id = smart_taxes setting_id = estates item = 1 }
	}
	if = { limit = { country_has_estate = estate_type:clergy_estate }
		cmm_show_list_item = { mod_id = smart_taxes setting_id = estates item = 2 }
	} else = {
		cmm_hide_list_item = { mod_id = smart_taxes setting_id = estates item = 2 }
	}
	if = { limit = { country_has_estate = estate_type:burghers_estate }
		cmm_show_list_item = { mod_id = smart_taxes setting_id = estates item = 3 }
	} else = {
		cmm_hide_list_item = { mod_id = smart_taxes setting_id = estates item = 3 }
	}
	if = { limit = { country_has_estate = estate_type:peasants_estate }
		cmm_show_list_item = { mod_id = smart_taxes setting_id = estates item = 4 }
	} else = {
		cmm_hide_list_item = { mod_id = smart_taxes setting_id = estates item = 4 }
	}
	if = { limit = { country_has_estate = estate_type:tribes_estate }
		cmm_show_list_item = { mod_id = smart_taxes setting_id = estates item = 5 }
	} else = {
		cmm_hide_list_item = { mod_id = smart_taxes setting_id = estates item = 5 }
	}
	if = { limit = { country_has_estate = estate_type:cossacks_estate }
		cmm_show_list_item = { mod_id = smart_taxes setting_id = estates item = 6 }
	} else = {
		cmm_hide_list_item = { mod_id = smart_taxes setting_id = estates item = 6 }
	}
	if = { limit = { country_has_estate = estate_type:dhimmi_estate }
		cmm_show_list_item = { mod_id = smart_taxes setting_id = estates item = 7 }
	} else = {
		cmm_hide_list_item = { mod_id = smart_taxes setting_id = estates item = 7 }
	}

	# Sync all CMM values to country variables on registration (handles save load)
	smart_taxes_sync_all = yes
}
