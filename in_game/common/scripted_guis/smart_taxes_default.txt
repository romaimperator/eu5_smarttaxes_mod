smart_taxes_setup_defaults = {
	effect = {
		# debug_log = "Smart Taxes: Setup Defaults"
		scope:country = {
			if = { limit = { not = { exists = var:nobles_estate_target } }
				set_variable = { name = nobles_estate_target value = 0.5 }
				set_variable = { name = smart_taxes_recalculate value = yes }
			}
			if = { limit = { not = { exists = var:clergy_estate_target } }
				set_variable = { name = clergy_estate_target value = 0.5 }
				set_variable = { name = smart_taxes_recalculate value = yes }
			}
			if = { limit = { not = { exists = var:burghers_estate_target } }
				set_variable = { name = burghers_estate_target value = 0.5 }
				set_variable = { name = smart_taxes_recalculate value = yes }
			}
			if = { limit = { not = { exists = var:peasants_estate_target } }
				set_variable = { name = peasants_estate_target value = 0.5 }
				set_variable = { name = smart_taxes_recalculate value = yes }
			}
			if = { limit = { not = { exists = var:tribes_estate_target } }
				set_variable = { name = tribes_estate_target value = 0.5 }
				set_variable = { name = smart_taxes_recalculate value = yes }
			}
			if = { limit = { not = { exists = var:cossacks_estate_target } }
				set_variable = { name = cossacks_estate_target value = 0.5 }
				set_variable = { name = smart_taxes_recalculate value = yes }
			}
			if = { limit = { not = { exists = var:dhimmi_estate_target } }
				set_variable = { name = dhimmi_estate_target value = 0.5 }
				set_variable = { name = smart_taxes_recalculate value = yes }
			}
		}
	}
}

smart_taxes_show_topbar_to_recompute = {
	is_shown = {
		exists = var:smart_taxes_recalculate
	}
}

smart_taxes_topbar_hide = {
	effect = {
		remove_variable = smart_taxes_recalculate
	}
}

smart_taxes_enabled = {
	is_shown = {
		OR = {
			AND = { scope:estate_type = estate_type:nobles_estate not = { exists = root.var:nobles_estate_disabled } }
			AND = { scope:estate_type = estate_type:clergy_estate not = { exists = root.var:clergy_estate_disabled } }
			AND = { scope:estate_type = estate_type:burghers_estate not = { exists = root.var:burghers_estate_disabled } }
			AND = { scope:estate_type = estate_type:peasants_estate not = { exists = root.var:peasants_estate_disabled } }
			AND = { scope:estate_type = estate_type:tribes_estate not = { exists = root.var:tribes_estate_disabled } }
			AND = { scope:estate_type = estate_type:cossacks_estate not = { exists = root.var:cossacks_estate_disabled } }
			AND = { scope:estate_type = estate_type:dhimmi_estate not = { exists = root.var:dhimmi_estate_disabled } }
		}
	}
}
