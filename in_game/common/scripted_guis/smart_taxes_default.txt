smart_taxes_should_recalculate = {
	is_shown = {
		exists = var:smart_taxes_recalculate
	}
}

smart_taxes_clear_recalculate = {
	effect = {
		remove_variable = smart_taxes_recalculate
	}
}

smart_taxes_enabled = {
	saved_scopes = { estate_type }

	is_shown = {
		root = {
			is_target_in_variable_list = { name = smart_taxes_enabled_estates target = scope:estate_type }
		}
	}

	effect = {
		root = {
			if = {
				limit = { is_target_in_variable_list = { name = smart_taxes_enabled_estates target = scope:estate_type } }
				remove_list_variable = { name = smart_taxes_enabled_estates target = scope:estate_type }
			} else = {
				add_to_variable_list = { name = smart_taxes_enabled_estates target = scope:estate_type }
			}
			smart_taxes_recompute_tax_rates = {}
		}
	}
}
