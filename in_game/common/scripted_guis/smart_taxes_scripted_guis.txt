smart_taxes_should_recalculate = {
	is_shown = {
		exists = var:smart_taxes_recalculate
	}
}

smart_taxes_clear_recalculate = {
	effect = {
		remove_variable = smart_taxes_recalculate
	}
}

# Called from the economy panel auto-tax checkbox override.
# Scope: country. Requires scope:estate_type.
smart_taxes_toggle = {
	scope = country
	effect = {
		smart_taxes_toggle_estate = yes
	}
}

smart_taxes_enabled = {
	is_shown = {
		root = {
			is_target_in_variable_list = { name = smart_taxes_enabled_estates target = scope:estate_type }
		}
	}
}
