# Hook this mod into CMF shared registration on_action.
cmf_on_mod_registration = {
	on_actions = {
		smart_taxes_on_register_mod
	}
}

smart_taxes_on_register_mod = {
	effect = {
		smart_taxes_register_mod = yes
	}
}

# Unified callback hook — handles setting changes.
cmf_on_callback = {
	on_actions = {
		smart_taxes_on_callback
	}
}

smart_taxes_on_callback = {
	effect = {
		smart_taxes_sync_all = yes
	}
}

# Recalculation via CMF shared pulse (already filtered to human players).
# Delays 28 days so taxes update right before the next month tick.
cmf_monthly_human_country_pulse = {
	on_actions = {
		smart_taxes_on_monthly
	}
}

smart_taxes_on_monthly = {
	on_actions = {
		delay = { days = 28 }
		smart_taxes_set_recalculate
	}
}

smart_taxes_set_recalculate = {
	effect = {
		set_variable = { name = smart_taxes_recalculate value = yes }
	}
}
