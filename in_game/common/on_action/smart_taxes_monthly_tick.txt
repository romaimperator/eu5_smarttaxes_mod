monthly_country_pulse = {
	on_actions = {
		smart_taxes_monthly_tick
	}
}

# Recompute 28 days in so rates update right before taxes are collected each month.
smart_taxes_monthly_tick = {
	trigger = {
		is_ai = no
	}
	on_actions = {
		delay = { days = 28 }
		smart_taxes_apply_monthly_recompute
	}
}

smart_taxes_apply_monthly_recompute = {
	effect = {
		if = { limit = { debug_only = yes } debug_log = "Smart Taxes: running monthly recompute" }
		save_scope_as = country
		smart_taxes_setup_defaults = {}
		smart_taxes_recompute_tax_rates = {}
	}
}
